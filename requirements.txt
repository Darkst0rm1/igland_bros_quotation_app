# --- Security -------------------------------------------------------------
# AES-256-GCM, for the one secret the database has to carry: a queued
# invitation's capability URL. The portal otherwise stores only token hashes,
# and authenticated encryption under an environment-held key is what keeps that
# property true while an email waits in the outbox. Hand-rolling an AEAD from
# hashlib was the alternative and is not one.
cryptography==50.0.0
# bcrypt is used directly rather than via passlib: passlib 1.7.4 is unmaintained
# and probes bcrypt.__about__.__version__, which bcrypt >= 4.1 removed.
bcrypt==5.0.0

# --- Documents ------------------------------------------------------------
# Quotations export as either PDF or Word, chosen by the employee at download
# time. Both renderers consume the same document model so the two outputs
# cannot drift apart. Both are pure-Python wheels and work on Windows and on
# the Linux image Community Cloud runs.
reportlab==5.0.0
python-docx==1.2.0
# Pinned explicitly rather than relied on transitively via Streamlit: the logo
# upload uses it to decode, bomb-guard and re-encode untrusted images, which is
# a security control and must not depend on another package's dependency graph.
pillow==12.3.0

# --- Object storage (Community Cloud has an ephemeral filesystem) ---------
boto3==1.43.62                  # S3-compatible: Supabase Storage, R2, S3

# --- Testing --------------------------------------------------------------
pytest==9.1.1
# Used only by the tests, to read back generated PDFs and assert on their text
# — that the totals appear, that the DRAFT mark is present or absent, and that
# no cost or margin figure leaked into a customer document.
pypdf==6.14.2
# Also test-only: parses render.yaml so the deployment Blueprint is checked by
# the suite rather than by a failed deploy — start commands, health path, and
# the assertion that no credential was ever written into it.
PyYAML==6.0.3

# --- Customer portal (separate public service) ----------------------------
# starlette, jinja2 and pydantic already arrive with Streamlit; only FastAPI is
# additional. httpx is needed by Starlette's TestClient.
fastapi==0.141.1
httpx==0.28.1
# Pinned explicitly rather than relied on transitively via Streamlit. The portal
# is deployed as its own service and its start command *is* the uvicorn CLI —
# an undeclared dependency there is a start-up failure, not a missing feature.
# It also means the portal host never needs Streamlit installed to serve.
uvicorn[standard]==0.52.1
